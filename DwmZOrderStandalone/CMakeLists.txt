cmake_minimum_required(VERSION 3.24)
project(DwmOrderTool VERSION 1.0.0 LANGUAGES CXX ASM_MASM RC)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_GENERATOR_PLATFORM MATCHES "ARM")
  message(FATAL_ERROR "Use MSVC with the x64 platform (Visual Studio 2022 / Build Tools).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/core/DwmZOrderAgent/RuntimeResolver.cpp")
  set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/core")
else()
  get_filename_component(CORE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
endif()
set(AGENT "${CORE}/DwmZOrderAgent")
set(CLIENT "${CORE}/Ksword5.1/Ksword5.1/OtherDock")
function(configure_native target)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
  target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/utf-8;/guard:cf>")
  target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
  target_include_directories(${target} PRIVATE "${AGENT}" "${CLIENT}" "${CORE}/shared/window")
endfunction()
add_library(KswordDwmZOrder SHARED "${AGENT}/DwmZOrderAgent.cpp" "${AGENT}/RuntimeResolver.cpp"
  "${AGENT}/RuntimeResolver.h" "${AGENT}/RuntimeSignatures.h" "${AGENT}/RuntimeWin10Signatures.h"
  "${AGENT}/NativeQueries.h" "${AGENT}/OrderPlan.h")
configure_native(KswordDwmZOrder)
target_link_libraries(KswordDwmZOrder PRIVATE user32 advapi32)
target_link_options(KswordDwmZOrder PRIVATE /MANIFESTUAC:NO)
add_library(DwmClient STATIC "${CLIENT}/DwmZOrderClient.cpp" "${CLIENT}/DwmAgentDeployment.cpp"
  "${CLIENT}/DwmRemoteLoader.asm" "${CLIENT}/DwmZOrderClient.h" "${CLIENT}/DwmAgentDeployment.h"
  "${CORE}/shared/window/DwmZOrderProtocol.h" "${CORE}/shared/window/DwmRemoteLoader.h"
  "${CORE}/shared/window/DwmProcessIdentity.h")
configure_native(DwmClient)
target_link_libraries(DwmClient PUBLIC user32 advapi32)
add_executable(DwmOrderTool WIN32 Main.cpp Diagnostics.cpp Diagnostics.h app.rc)
configure_native(DwmOrderTool)
target_link_libraries(DwmOrderTool PRIVATE DwmClient comctl32 comdlg32 shell32 version)
target_link_options(DwmOrderTool PRIVATE /MANIFEST:NO)
target_sources(DwmOrderTool PRIVATE "${AGENT}/RuntimeResolver.cpp")
add_dependencies(DwmOrderTool KswordDwmZOrder)
source_group("Application" FILES Main.cpp Diagnostics.cpp Diagnostics.h app.rc app.manifest)
include(CTest)
if(BUILD_TESTING)
  add_library(DwmLoaderFixture SHARED "${AGENT}/tests/LoaderFixture.cpp" "${AGENT}/tests/PrivateQueryFixture.cpp")
  configure_native(DwmLoaderFixture)
  target_link_libraries(DwmLoaderFixture PRIVATE advapi32)
  target_link_options(DwmLoaderFixture PRIVATE /MANIFESTUAC:NO)
  add_custom_command(TARGET DwmLoaderFixture POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "$<TARGET_FILE:DwmLoaderFixture>" "$<TARGET_FILE_DIR:DwmLoaderFixture>/DwmLoaderReject.dll")
  add_executable(DwmZOrderTests "${AGENT}/tests/OrderTests.cpp" "${AGENT}/tests/LoaderTests.cpp"
    "${AGENT}/tests/CfgTests.cpp" "${AGENT}/tests/DeploymentTests.cpp" "${AGENT}/tests/RuntimeTests.cpp"
    "${AGENT}/RuntimeResolver.cpp")
  configure_native(DwmZOrderTests)
  target_link_libraries(DwmZOrderTests PRIVATE DwmClient)
  add_dependencies(DwmZOrderTests DwmLoaderFixture KswordDwmZOrder)
  add_test(NAME core-regression COMMAND DwmZOrderTests "$<TARGET_FILE:KswordDwmZOrder>")
  add_test(NAME diagnostic-report COMMAND DwmOrderTool --self-test "${CMAKE_BINARY_DIR}/diagnostic-test")
  set_tests_properties(core-regression diagnostic-report PROPERTIES TIMEOUT 90)
endif()
