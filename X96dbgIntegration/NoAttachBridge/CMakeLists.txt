cmake_minimum_required(VERSION 3.20)
project(KSwordNoAttach LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The independent HVM backend requires native AMD64")
endif()
add_library(KSwordNoAttach SHARED NoAttachBridge.cpp)
target_compile_features(KSwordNoAttach PRIVATE cxx_std_17)
target_compile_definitions(KSwordNoAttach PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(MSVC)
  target_compile_options(KSwordNoAttach PRIVATE /W4 /WX /utf-8)
  set_property(TARGET KSwordNoAttach PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set_target_properties(KSwordNoAttach PROPERTIES PREFIX "" SUFFIX ".dp64")
